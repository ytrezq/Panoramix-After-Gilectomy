def SafeMath:
  def add(a, b):
      if a + b < a:
          revert with 'ds-math-add-overflow'
      return a + b
  def sub(a, b):
      if a - b > a:
          revert with 'ds-math-sub-underflow'
      return a - b
  def mul(a, b):
      if a * b / a != b:
          revert with 'ds-math-mul-overflow'
      return a * b
  def div(a, b):
      assert b
      return a / b
