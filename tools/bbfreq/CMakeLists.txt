cmake_minimum_required(VERSION 3.14)
project(bbfreq C)
find_package(DynamoRIO)
if (NOT DynamoRIO_FOUND)
  message(FATAL_ERROR "DynamoRIO package required to build")
endif()
add_library(bbfreq SHARED bbfreq.c)
configure_DynamoRIO_client(bbfreq)
use_DynamoRIO_extension(bbfreq drmgr)
use_DynamoRIO_extension(bbfreq drx)
use_DynamoRIO_extension(bbfreq drcontainers)
use_DynamoRIO_extension(bbfreq drreg)
